    70→/** 文件解析结果 → 提交后端的归一输入串（优先三位 ID，其次 名称+分子式） */
    71→function ligandRawFromParsed(p: LigandFileParsed, fileName: string): string {
    72→  if (p.compId) return p.compId
    73→  const parts = [p.name, p.formula].filter(Boolean)
    74→  if (parts.length > 0) return parts.join(' ')
    75→  return fileName.replace(/\.[^.]+$/, '').trim().slice(0, 60)
    76→}

   148→/** 输入视图（idle）：一屏工作台卡片 — 步骤轨道（靶点 / 配体 / 目标）+ 高级选项 + 提交条 */
   149→export function InputPanel({ submitting, submitError, onStart, onClearSubmitError }: InputPanelProps) {
   150→  const [mode, setMode] = useState<TabMode>('uniprot')
   151→
   152→  // UniProt 输入（v1.22：支持多 ID，逗号/空格/换行分隔）
   153→  const [uniprot, setUniprot] = useState('')
   154→  const uniprotParsed = useMemo(() => parseMultiUniprotIds(uniprot), [uniprot])
   155→  const uniprotValid = uniprotParsed.valid.length >= 1 && uniprotParsed.invalid.length === 0
   156→  const uniprotMulti = uniprotParsed.valid.length > 1
   157→  const uniprotOverLimit = uniprotParsed.valid.length > MAX_BATCH_UNIPROT
   158→
   159→  // UniProt Tab 的可选伴随序列（合并评估）
   160→  const [companionOpen, setCompanionOpen] = useState(false)
   161→  const [companionText, setCompanionText] = useState('')
   162→  const companionInfo = useMemo(() => countResidues(companionText), [companionText])
   163→  const companionValid = companionInfo.count >= MIN_COMPANION_LENGTH
   164→
   165→  // v1.23 评估范围限定（UniProt 残基编号，1-based 闭区间）— 仅 UniProt 单靶点无伴随序列时启用
   166→  const [rangeOpen, setRangeOpen] = useState(false)
   167→  const [rangeStart, setRangeStart] = useState('')
   168→  const [rangeEnd, setRangeEnd] = useState('')
   169→  const rangeStartN = parseInt(rangeStart, 10)
   170→  const rangeEndN = parseInt(rangeEnd, 10)
   171→  const rangeValid = Number.isFinite(rangeStartN) && Number.isFinite(rangeEndN)
   172→    && rangeStartN >= 1 && rangeEndN >= rangeStartN && rangeEndN <= 100000
   173→  // 多靶点或有伴随序列时不启用范围（语义不适用：多靶点各自范围不同；伴随序列已隐含构建边界）
   174→  const rangeEnabled = uniprotParsed.valid.length === 1 && !companionValid
   175→  const range = rangeEnabled && rangeValid ? [rangeStartN, rangeEndN] as [number, number] : undefined
   176→
   177→  // 序列输入（v1.22：支持多 FASTA，每个 `>` 标题段一条）
   178→  const [sequenceText, setSequenceText] = useState('')
   179→  const seqInfo = useMemo(() => countResidues(sequenceText), [sequenceText])
   180→  const seqEntries = useMemo(() => parseMultiFasta(sequenceText), [sequenceText])
   181→  const seqMulti = seqEntries.length > 1
   182→  const seqValidEntries = seqEntries.filter(e => e.length >= MIN_SEQUENCE_LENGTH)
   183→  const seqOverLimit = seqEntries.length > MAX_BATCH_UNIPROT
   184→
   185→  // Word 评估表上传
   186→  const [upload, setUpload] = useState<UploadResponse | null>(null)
   187→  const [uploading, setUploading] = useState(false)
   188→  const [uploadProgress, setUploadProgress] = useState<UploadProgress | null>(null)
   189→  const [uploadError, setUploadError] = useState<string | null>(null)
   190→  /** 已选 UniProt 靶点（多选：每个靶点生成一份独立评估报告） */
   191→  const [selectedUniprots, setSelectedUniprots] = useState<string[]>([])
   192→  const [selectedSequenceIndex, setSelectedSequenceIndex] = useState<number | null>(null)
   193→
   194→  // AI 智能解析
   195→  const [insights, setInsights] = useState<FormInsights | null>(null)
   196→  const [insightsLoading, setInsightsLoading] = useState(false)
   197→  const [insightsError, setInsightsError] = useState<string | null>(null)
   198→  const [goalAutoFilled, setGoalAutoFilled] = useState(false)
   199→
   200→  // v2.0 配体输入（文本草稿 + 芯片；文件经 ligand-parse 客户端解析）
   201→  const [ligandDraft, setLigandDraft] = useState('')
   202→  const [ligandChips, setLigandChips] = useState<LigandChip[]>([])
   203→  const ligandFileInputRef = useRef<HTMLInputElement>(null)
   204→  const ligandSeqRef = useRef(1)
   205→
   206→  const docxReady = Boolean(
   207→    upload && (selectedUniprots.length > 0 || (selectedSequenceIndex != null && upload.parsed.sequences[selectedSequenceIndex]))
   208→  )
   209→
   210→  // 客户目标
   211→  const [goal, setGoal] = useState('')
   212→
   213→  // 高级选项
   214→  const [advOpen, setAdvOpen] = useState(false)
   215→  const [blastEnabled, setBlastEnabled] = useState(true)
   216→  const [forceBlast, setForceBlast] = useState(false)
   217→  const [maxPdb, setMaxPdb] = useState(150)
   218→
   219→  // 快捷键提示文案（挂载后按平台修正，避免 SSR 水合不一致）
   220→  const [kbdHint, setKbdHint] = useState('⌘ ↵')
   221→  useEffect(() => {
   222→    if (typeof navigator !== 'undefined' && !/Mac|iPhone|iPad/.test(navigator.platform ?? navigator.userAgent)) {
   223→      setKbdHint('Ctrl ↵')
   224→    }
   225→  }, [])
   226→
   227→  /** 上传成功后自动执行 LLM 智能解析（不阻塞选择评估对象） */
   228→  const runInsights = async (uploadId: string) => {
   229→    setInsightsLoading(true)
   230→    setInsightsError(null)
   231→    setInsights(null)
   232→    try {
   233→      const res = await fetchDocxInsights(uploadId)
   234→      const data = res.insights
   235→      if (!data) throw new Error('解析结果为空')
   236→      setInsights(data)
   237→      // 目标输入框为空时自动填入解析出的客户目标
   238→      if (data.goals.length > 0 && goal.trim() === '') {
   239→        setGoal(data.goals.slice(0, 3).join('；'))
   240→        setGoalAutoFilled(true)
   241→      }
   242→    } catch (e) {
   243→      setInsightsError(e instanceof Error ? e.message : '智能解析失败')
   244→    } finally {
   245→      setInsightsLoading(false)
   246→    }
   247→  }
   248→
   249→  const handleFileSelected = async (file: File) => {
   250→    const invalid = validateDocxFile(file)
   251→    if (invalid) {
   252→      setUploadError(invalid)
   253→      return
   254→    }
   255→    setUploading(true)
   256→    setUploadProgress(null)
   257→    setUploadError(null)
   258→    setInsights(null)
   259→    setInsightsError(null)
   260→    try {
   261→      const res = await uploadEvaluateDocx(file, setUploadProgress)
   262→      setUpload(res)
   263→      // 默认选中全部 UniProt（用户要求：表中的 UniProt 都要评估，不落下；上限 MAX_BATCH_UNIPROT 个）
   264→      if (res.parsed.uniprotIds.length > 0) {
   265→        setSelectedUniprots(res.parsed.uniprotIds.slice(0, MAX_BATCH_UNIPROT))
   266→        setSelectedSequenceIndex(null)
   267→      } else if (res.parsed.sequences.length > 0) {
   268→        setSelectedUniprots([])
   269→        setSelectedSequenceIndex(0)
   270→      } else {
   271→        setSelectedUniprots([])
   272→        setSelectedSequenceIndex(null)
   273→      }
   274→      // 后台执行 LLM 智能解析
   275→      void runInsights(res.uploadId)
   276→    } catch (err) {
   277→      setUploadError(err instanceof Error ? err.message : '上传解析失败，请稍后重试')
   278→    } finally {
   279→      setUploading(false)
   280→      setUploadProgress(null)
   281→    }
   282→  }
   283→
   284→  const clearUpload = () => {
   285→    setUpload(null)
   286→    setSelectedUniprots([])
   287→    setSelectedSequenceIndex(null)
   288→    setUploadError(null)
   289→    setUploadProgress(null)
   290→    setInsights(null)
   291→    setInsightsError(null)
   292→  }
   293→
   294→  const applyGoal = (text: string) => {
   295→    setGoal(text)
   296→    setGoalAutoFilled(false)
   297→  }
   298→
   299→  const inputReady =
   300→    mode === 'uniprot'
   301→      ? uniprotValid && !uniprotOverLimit
   302→      : mode === 'sequence'
   303→        ? seqInfo.count >= MIN_SEQUENCE_LENGTH && !seqOverLimit
   304→        : docxReady
   305→
   306→  /** 切换 UniProt 靶点选中态（多选，上限 MAX_BATCH_UNIPROT） */
   307→  const toggleUniprot = (id: string) => {
   308→    setSelectedUniprots(prev =>
   309→      prev.includes(id) ? prev.filter(x => x !== id) : prev.length < MAX_BATCH_UNIPROT ? [...prev, id] : prev,
   310→    )
   311→  }
   312→
   313→  // ─── 配体芯片操作 ───────────────────────────────────────────────────────────
   314→
   315→  const addLigandChip = (raw: string, source: 'text' | 'file', fileName?: string, parsed?: LigandFileParsed) => {
   316→    const v = raw.trim().slice(0, 160)
   317→    if (!v) return
   318→    setLigandChips(prev => {
   319→      if (prev.some(c => c.raw.toLowerCase() === v.toLowerCase())) return prev
   320→      if (prev.length >= MAX_LIGANDS) return prev
   321→      return [...prev, { id: ligandSeqRef.current++, raw: v, source, fileName, parsed }]
   322→    })
   323→  }
   324→
   325→  const commitLigandDraft = () => {
   326→    const v = ligandDraft.trim()
   327→    if (!v) return
   328→    addLigandChip(v, 'text')
   329→    setLigandDraft('')
   330→  }
   331→
   332→  const removeLigandChip = (id: number) => {
   333→    setLigandChips(prev => prev.filter(c => c.id !== id))
   334→  }
   335→
   336→  const handleLigandFile = async (file: File) => {
   337→    if (file.size > 2 * 1024 * 1024) return
   338→    try {
   339→      const text = await file.text()
   340→      const parsed = parseLigandFileText(file.name, text)
   341→      addLigandChip(ligandRawFromParsed(parsed, file.name), 'file', file.name, parsed.kind !== 'unknown' ? parsed : undefined)
   342→    } catch {
   343→      addLigandChip(file.name.replace(/\.[^.]+$/, '').slice(0, 60), 'file', file.name)
   344→    }
   345→  }
   346→
   347→  const handleStart = async () => {
   348→    if (!inputReady || submitting) return
   349→    onClearSubmitError()
   350→    const options: EvaluateOptions = {
   351→      skipBlast: !blastEnabled,
   352→      forceBlast: blastEnabled && forceBlast,
   353→      maxPdb,
   354→    }
   355→    const ligands = ligandChips.length
   356→      ? ligandChips.map(c => ({ raw: c.raw, source: c.source, fileName: c.fileName }))
   357→      : undefined
   358→    const trimmedGoal = goal.trim() || undefined
   359→    let requests: EvaluateRequest[]
   360→    if (mode === 'uniprot') {
   361→      // v1.22：支持多 UniProt ID — 每个 ID 一份独立作业（批量编排 + 关系判断 + 共有结构）。
   362→      // 单 ID 时保留可选伴随序列（同靶点合并评估）；多 ID 时伴随序列不附加（语义不适用）。
   363→      // v1.23：单 ID 无伴随序列时支持评估范围限定（range）。
   364→      const ids = uniprotParsed.valid.slice(0, MAX_BATCH_UNIPROT)
   365→      const companion = companionText.trim() && companionValid && ids.length === 1 ? companionText.trim() : undefined
   366→      const reqRange = ids.length === 1 && !companion ? range : undefined
   367→      requests = ids.map(id => ({
   368→        mode: 'uniprot' as const,
   369→        uniprot: id,
   370→        sequence: companion,
   371→        goal: trimmedGoal,
   372→        ligands,
   373→        options: { ...options, range: reqRange },
   374→      }))
   375→    } else if (mode === 'sequence') {
   376→      // v1.22：支持多 FASTA — 每条序列一份独立作业（批量编排 + 关系判断 + 共有结构）。
   377→      // 单条序列直接提交；多条按 `>` 标题段拆分，每条 ≥MIN_SEQUENCE_LENGTH 的独立评估。
   378→      const entries = seqEntries.slice(0, MAX_BATCH_UNIPROT)
   379→      if (entries.length === 0) return
   380→      requests = entries.map(e => ({
   381→        mode: 'sequence' as const,
   382→        sequence: e.seq,
   383→        goal: trimmedGoal,
   384→        ligands,
   385→        options,
   386→      }))
   387→    } else if (!upload) {
   388→      return
   389→    } else {
   390→      const selectedSeq =
   391→        selectedSequenceIndex != null
   392→          ? (upload.parsed.sequences[selectedSequenceIndex]?.seq ?? undefined)
   393→          : undefined
   394→      // 多 UniProt 靶点 → 每个靶点一份独立作业（串行执行，全部不落下）；
   395→      // 仅选序列时 → 单作业。伴随序列随每个作业提交，由后端同靶点校验决定是否合并；
   396→      // v1.10：评估表全部序列一并上送 — 关系判断归属后（同源家族/构建归属）按靶点附构建序列
   397→      const targets = selectedUniprots.length > 0 ? selectedUniprots : [null]
   398→      requests = targets.map(u => ({
   399→        mode: 'docx' as const,
   400→        // 契约字段 docxJobId + 兼容别名 uploadId（同值），后端取任一均可
   401→        docxJobId: upload.uploadId,
   402→        uploadId: upload.uploadId,
   403→        uniprotFromDocx: u ?? undefined,
   404→        sequence: selectedSeq,
   405→        sequenceFromDocx: selectedSeq,
   406→        goal: trimmedGoal,
   407→        ligands,
   408→        options,
   409→      }))
   410→      const docxSequences = upload.parsed.sequences.map(s => s.seq ?? '')
   411→      await onStart(requests, { docxSequences })
   412→      return
   413→    }
   414→    await onStart(requests)
   415→  }
   416→
   417→  // ⌘/Ctrl + Enter 快捷开始评估（不设依赖数组：始终绑定最新闭包）
   418→  useEffect(() => {
   419→    const onKey = (e: KeyboardEvent) => {
   420→      if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
   421→        e.preventDefault()
   422→        if (inputReady && !submitting) void handleStart()
   423→      }
   424→    }
   425→    window.addEventListener('keydown', onKey)
   426→    return () => window.removeEventListener('keydown', onKey)
   427→  })
